#include "DHT.h"
#define DHTPIN 2      // DHT11 信号脚
#define LED 12        // LED
#define DHTTYPE DHT11
#define WORRING 11    // 蜂鸣器
#define LIGHT_PIN A3  // 光敏电阻接 A3

DHT dht(DHTPIN, DHTTYPE);

void setup() {
  Serial.begin(9600);
  dht.begin();
  pinMode(LED, OUTPUT);
  pinMode(WORRING, OUTPUT);
}

void loop() {
  // 读取温湿度
  float humidity = dht.readHumidity();
  float temperature = dht.readTemperature();

  // 读取光敏电阻（A3）
  int lightValue = analogRead(LIGHT_PIN);

  // 温湿度读取失败 → 蜂鸣器报警
  if (isnan(humidity) || isnan(temperature)||lightValue==0) {
    Serial.println("error01");
    digitalWrite(LED,LOW);
    digitalWrite(WORRING, HIGH);
    delay(200);
    digitalWrite(WORRING, LOW);
    delay(200);
    return;
  }

  // 串口输出：湿度 温度 光强
  Serial.print(humidity);
  Serial.print(" ");
  Serial.print(temperature);
  Serial.print(" ");
  Serial.println(lightValue);  // 这里输出光敏值

  if(lightValue<=400){
    digitalWrite(LED,HIGH);
    delay(2000);
    return;
  }
  digitalWrite(LED,LOW);
  delay (1000);
}
